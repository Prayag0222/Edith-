import subprocess
from dataclasses import dataclass

import ctranslate2


@dataclass
class VoiceDecision:
    model_size: str
    device: str
    compute_type: str
    reason: str


class AdaptiveVoiceEngine:
    MIN_FREE_VRAM_MB = 1800.0

    def __init__(self):
        self.gpu_available = self._check_gpu()

    @staticmethod
    def _check_gpu() -> bool:
        try:
            return ctranslate2.get_cuda_device_count() > 0
        except Exception:
            return False

    def get_gpu_stats(self) -> dict:
        fallback = {
            "gpu_usage": 100.0,
            "vram_used": 0.0,
            "vram_total": 0.0,
            "vram_free": 0.0,
            "vram_usage": 100.0,
            "available": False,
        }

        try:
            result = subprocess.run(
                [
                    "nvidia-smi",
                    "--query-gpu=utilization.gpu,memory.used,memory.total",
                    "--format=csv,noheader,nounits",
                ],
                capture_output=True,
                text=True,
                timeout=2,
                check=False,
            )

            if result.returncode != 0:
                return fallback

            line = result.stdout.strip().splitlines()[0]
            gpu_usage, vram_used, vram_total = (
                float(value.strip())
                for value in line.split(",")
            )

            if vram_total <= 0:
                return fallback

            vram_free = max(0.0, vram_total - vram_used)
            vram_usage = (vram_used / vram_total) * 100.0

            return {
                "gpu_usage": gpu_usage,
                "vram_used": vram_used,
                "vram_total": vram_total,
                "vram_free": vram_free,
                "vram_usage": vram_usage,
                "available": True,
            }

        except (OSError, ValueError, IndexError, subprocess.SubprocessError):
            return fallback

    def choose_model(
        self,
        task_type: str = "conversation",
        gpu_model_loaded: bool = False,
    ) -> VoiceDecision:
        if not self.gpu_available:
            return VoiceDecision(
                "small", "cpu", "int8", "GPU unavailable"
            )

        if gpu_model_loaded:
            stats = self.get_gpu_stats()
            gpu_usage = stats.get("gpu_usage")
            reason = f"Reusing cached GPU model for {task_type}"
            if gpu_usage is not None:
                reason += f" (GPU {gpu_usage:.0f}%)"
            return VoiceDecision(
                "medium", "cuda", "float16",
                reason,
            )

        stats = self.get_gpu_stats()

        if not stats["available"]:
            return VoiceDecision(
                "small", "cpu", "int8",
                "GPU resource monitoring unavailable",
            )

        gpu_usage = stats["gpu_usage"]
        vram_free = stats["vram_free"]

        if vram_free < self.MIN_FREE_VRAM_MB:
            return VoiceDecision(
                "small", "cpu", "int8",
                f"Insufficient free VRAM ({vram_free:.0f} MB)",
            )

        return VoiceDecision(
            "medium", "cuda", "float16",
            f"GPU prioritized for {task_type} "
            f"(GPU {gpu_usage:.0f}%, free VRAM {vram_free:.0f} MB)",
        )

    def get_decision(
        self,
        task_type: str = "conversation",
        gpu_model_loaded: bool = False,
    ) -> VoiceDecision:
        return self.choose_model(
            task_type=task_type,
            gpu_model_loaded=gpu_model_loaded,
        )

import tempfile
import wave
from collections import deque

import numpy as np
import sounddevice as sd

from app.voice.vad import VoiceActivityDetector


class AudioRecorder:
    """
    Handles microphone input and audio preparation.

    Responsibilities:
        - Capture microphone audio
        - Detect human speech using WebRTC VAD
        - Reject silence/background noise
        - Keep a short pre-buffer
        - Continue recording through brief VAD mistakes
        - Detect genuine end-of-speech
        - Convert recorded audio into a WAV file
    """

    def __init__(
        self,
        sample_rate: int = 16000,
        channels: int = 1,
        silence_duration: float = 1.5,
        max_recording_seconds: float = 20.0,
        vad_aggressiveness: int = 2,
        initial_silence_timeout: float = 8.0,

    ):
        self.sample_rate = sample_rate
        self.channels = channels

        self.silence_duration = silence_duration
        self.max_recording_seconds = (
            max_recording_seconds
        )
        self.initial_silence_timeout = initial_silence_timeout

        self.vad = VoiceActivityDetector(
            sample_rate=self.sample_rate,
            aggressiveness=vad_aggressiveness,
        )

    # --------------------------------------------------
    # AUDIO CONVERSION
    # --------------------------------------------------

    def _float_to_pcm16(
        self,
        audio: np.ndarray,
    ) -> bytes:
        """
        Convert float32 microphone audio into
        signed 16-bit PCM bytes.

        WebRTC VAD requires:
            - mono
            - PCM16
            - 10/20/30 ms frames
        """

        audio = np.asarray(
            audio,
            dtype=np.float32,
        )

        audio = np.clip(
            audio,
            -1.0,
            1.0,
        )

        pcm_audio = (
            audio * 32767
        ).astype(
            np.int16
        )

        return pcm_audio.tobytes()

    # --------------------------------------------------
    # AUDIO ENERGY
    # --------------------------------------------------

    def _calculate_rms(
        self,
        audio: np.ndarray,
    ) -> float:
        """
        Calculate RMS audio energy.

        RMS gives us a simple measurement of
        how loud the microphone signal is.
        """

        if len(audio) == 0:
            return 0.0

        audio = audio.astype(
            np.float32
        )

        return float(
            np.sqrt(
                np.mean(
                    np.square(audio)
                )
            )
        )

    # --------------------------------------------------
    # FRAME SPLITTING
    # --------------------------------------------------

    def _frame_audio(
        self,
        audio: np.ndarray,
        frame_duration_ms: int = 30,
    ):
        """
        Split audio into WebRTC-compatible frames.

        WebRTC VAD supports:
            10 ms
            20 ms
            30 ms

        EDITH uses 30 ms frames.
        """

        frame_size = int(
            self.sample_rate
            * frame_duration_ms
            / 1000
        )

        total_samples = len(audio)

        for start in range(
            0,
            total_samples - frame_size + 1,
            frame_size,
        ):
            end = start + frame_size

            yield audio[start:end]

    # --------------------------------------------------
    # VAD ANALYSIS
    # --------------------------------------------------

    def _analyze_speech(
        self,
        audio_chunk: np.ndarray,
    ):
        """
        Analyze one microphone block.

        Returns:

            {
                "rms": float,
                "speech_frames": int,
                "total_frames": int,
            }

        This method does NOT decide whether EDITH
        should start or stop recording.

        It only gathers evidence.
        """

        rms = self._calculate_rms(
            audio_chunk
        )

        speech_frames = 0
        total_frames = 0

        for frame in self._frame_audio(
            audio_chunk,
            frame_duration_ms=30,
        ):
            pcm_frame = (
                self._float_to_pcm16(
                    frame
                )
            )

            try:

                is_speech = (
                    self.vad.is_speech(
                        pcm_frame
                    )
                )

            except Exception:

                is_speech = False

            total_frames += 1

            if is_speech:
                speech_frames += 1

        return {
            "rms": rms,
            "speech_frames": speech_frames,
            "total_frames": total_frames,
        }

    # --------------------------------------------------
    # SPEECH START DETECTION
    # --------------------------------------------------

    def _is_speech_start(
        self,
        analysis,
    ) -> bool:
        """
        Decide whether the user has actually started
        speaking.

        Starting recording is intentionally stricter
        than continuing an existing recording.

        This prevents random background noise from
        waking EDITH.
        """

        rms = analysis["rms"]
        speech_frames = analysis[
            "speech_frames"
        ]

        # Very quiet microphone signal.
        if rms < 0.008:
            return False

        # Require multiple VAD-positive frames
        # before declaring that speech has started.
        if speech_frames >= 2:
            return True

        return False

    # --------------------------------------------------
    # SPEECH CONTINUATION
    # --------------------------------------------------

    def _is_speech_continuation(
        self,
        analysis,
    ) -> bool:
        """
        Decide whether an already-active speech session
        should continue.

        This is intentionally more forgiving.

        Natural speech contains:
            - quiet syllables
            - pauses
            - consonants
            - short VAD mistakes

        We don't want those to terminate recording.
        """

        rms = analysis["rms"]
        speech_frames = analysis[
            "speech_frames"
        ]

        # Strong VAD evidence.
        if speech_frames >= 1:
            return True

        # Moderate audio energy.

        # This allows EDITH to survive short VAD
        # mistakes while the user is still talking.
        if rms >= 0.012:
            return True

        return False

    # --------------------------------------------------
    # RECORDING
    # --------------------------------------------------

    def record_until_silence(self):
        """
        Record microphone input until genuine silence.

        Recording has two states:

            WAITING
                ↓
            SPEECH START
                ↓
            RECORDING
                ↓
            TEMPORARY VAD MISS
                ↓
            CONTINUE
                ↓
            REAL SILENCE
                ↓
            STOP

        This prevents both:

            1. Background noise from triggering EDITH.
            2. Natural speech from being cut prematurely.
        """

        block_duration = 0.1

        block_size = int(
            self.sample_rate
            * block_duration
        )

        silence_blocks_required = max(
            1,
            int(
                self.silence_duration
                / block_duration
            ),
        )

        max_blocks = int(
            self.max_recording_seconds
            / block_duration
        )

        initial_silence_blocks = int(
            self.initial_silence_timeout
            / block_duration
        )

        # 500 ms pre-buffer.
        #
        # This protects the beginning of words.
        pre_buffer_blocks = 5

        pre_buffer = deque(
            maxlen=pre_buffer_blocks
        )

        recorded_chunks = []

        speaking = False
        silent_blocks = 0

        print("Listening...")

        with sd.InputStream(
            samplerate=self.sample_rate,
            channels=self.channels,
            dtype="float32",
            blocksize=block_size,
        ) as stream:

            for block_index in range(max_blocks):

                audio_chunk, overflowed = (
                    stream.read(
                        block_size
                    )
                )

                if overflowed:
                    print(
                        "Warning: microphone "
                        "buffer overflow."
                    )

                audio_chunk = (
                    audio_chunk.copy()
                )

                mono_chunk = (
                    audio_chunk[:, 0]
                )

                analysis = (
                    self._analyze_speech(
                        mono_chunk
                    )
                )

                pre_buffer.append(
                    mono_chunk
                )

                # ----------------------------------
                # WAITING FOR SPEECH
                # ----------------------------------

                if not speaking:

                    speech_started = (
                        self._is_speech_start(
                            analysis
                        )
                    )

                    if speech_started:

                        speaking = True

                        recorded_chunks.extend(
                            list(pre_buffer)
                        )

                        silent_blocks = 0

                    elif block_index + 1 >= initial_silence_blocks:
                        print("[STT] Initial silence timeout.")
                        return None


                    continue

                # ----------------------------------
                # ALREADY RECORDING
                # ----------------------------------

                speech_continues = (
                    self._is_speech_continuation(
                        analysis
                    )
                )

                recorded_chunks.append(
                    mono_chunk
                )

                if speech_continues:

                    # User is still talking.
                    #
                    # Reset the silence timer.
                    silent_blocks = 0

                else:

                    # Possible silence.
                    silent_blocks += 1

                    if (
                        silent_blocks
                        >= silence_blocks_required
                    ):
                        break

        # ------------------------------------------
        # NO SPEECH EVER STARTED
        # ------------------------------------------

        if not speaking:
            return None

        if not recorded_chunks:
            return None

        return np.concatenate(
            recorded_chunks
        )

    # --------------------------------------------------
    # WAV CREATION
    # --------------------------------------------------

    def save_wav(
        self,
        audio,
    ):
        """
        Save recorded audio as a temporary WAV file.
        """

        temporary_file = (
            tempfile.NamedTemporaryFile(
                suffix=".wav",
                delete=False,
            )
        )

        temporary_path = (
            temporary_file.name
        )

        temporary_file.close()

        audio = np.asarray(
            audio,
            dtype=np.float32,
        )

        audio = np.clip(
            audio,
            -1.0,
            1.0,
        )

        pcm_audio = (
            audio * 32767
        ).astype(
            np.int16
        )

        with wave.open(
            temporary_path,
            "wb",
        ) as wav_file:

            wav_file.setnchannels(
                self.channels
            )

            wav_file.setsampwidth(
                2
            )

            wav_file.setframerate(
                self.sample_rate
            )

            wav_file.writeframes(
                pcm_audio.tobytes()
            )

        return temporary_path

import os


def setup_cuda_runtime():
    try:
        import nvidia.cublas
        import nvidia.cudnn

        cublas_path = os.path.join(
            nvidia.cublas.__path__[0],
            "bin",
        )

        cudnn_path = os.path.join(
            nvidia.cudnn.__path__[0],
            "bin",
        )

        if hasattr(os, "add_dll_directory"):
            os.add_dll_directory(cublas_path)
            os.add_dll_directory(cudnn_path)

        os.environ["PATH"] = (
            cublas_path
            + os.pathsep
            + cudnn_path
            + os.pathsep
            + os.environ.get("PATH", "")
        )

        print("CUDA runtime libraries configured.")

        return True

    except Exception as e:
        print(f"CUDA runtime setup failed: {e}")
        return False

class WhisperModelManager:
    """
    Manages Whisper model loading and caching.

    Models:

        CPU:
            small + int8

        GPU:
            medium + float16

    Whisper and Faster-Whisper are imported lazily so that
    EDITH does not initialize the Whisper stack during startup.
    """

    def __init__(self):
        self.cpu_model = None
        self.gpu_model = None

    def get_cpu_model(self):
        """
        Return the cached CPU Whisper model.

        The model is loaded only when it is first requested.
        """

        if self.cpu_model is not None:
            return self.cpu_model

        from faster_whisper import WhisperModel

        print("[STT] Loading small model on CPU...")

        self.cpu_model = WhisperModel(
            "small",
            device="cpu",
            compute_type="int8",
            local_files_only=True,
        )

        print("[STT] CPU small model ready.")

        return self.cpu_model

    def get_gpu_model(self):
        """
        Return the cached GPU Whisper model.

        The model is loaded only when it is first requested.
        """

        if self.gpu_model is not None:
            return self.gpu_model

        from faster_whisper import WhisperModel

        print("[STT] Loading medium model on CUDA...")

        self.gpu_model = WhisperModel(
            "medium",
            device="cuda",
            compute_type="float16",
            local_files_only=True,
        )

        print("[STT] CUDA medium model ready.")

        return self.gpu_model

    def is_cpu_model_loaded(self) -> bool:
        """Return True if the CPU model is already loaded."""
        return self.cpu_model is not None

    def is_gpu_model_loaded(self) -> bool:
        """Return True if the GPU model is already loaded."""
        return self.gpu_model is not None

import numpy as np
import noisereduce as nr


class NoiseSuppressor:
    """
    Non-stationary spectral noise suppression.

    Responsibilities:
        - Reduce environmental background noise.
        - Preserve speech as much as possible.
        - Avoid unnecessary processing of short recordings.
        - Prevent excessive attenuation.
        - Keep the original audio if processing fails.
    """

    def __init__(
        self,
        sample_rate: int = 16000,
        strength: float = 0.75,
        enabled: bool = True,
        minimum_duration: float = 0.25,
    ):
        self.sample_rate = sample_rate
        self.strength = float(np.clip(strength, 0.0, 1.0))
        self.enabled = enabled
        self.minimum_duration = minimum_duration

    # -----------------------------------------------
    # AUDIO METRICS
    # -----------------------------------------------

    def _calculate_rms(self, audio: np.ndarray) -> float:
        if len(audio) == 0:
            return 0.0

        return float(
            np.sqrt(
                np.mean(
                    np.square(audio.astype(np.float32))
                )
            )
        )

    # -----------------------------------------------
    # NOISE SUPPRESSION
    # -----------------------------------------------

    def process(self, audio: np.ndarray) -> np.ndarray:

        audio = np.asarray(
            audio,
            dtype=np.float32,
        ).flatten()

        if not self.enabled:
            return audio

        if len(audio) == 0:
            return audio

        duration = len(audio) / self.sample_rate

        if duration < self.minimum_duration:
            return audio

        original_audio = audio.copy()

        original_rms = self._calculate_rms(
            original_audio
        )

        if original_rms < 0.001:
            return original_audio

        try:

            cleaned_audio = nr.reduce_noise(
                y=original_audio,
                sr=self.sample_rate,
                stationary=False,
                prop_decrease=self.strength,
                n_fft=512,
                hop_length=128,
            )

            cleaned_audio = np.asarray(
                cleaned_audio,
                dtype=np.float32,
            )

            cleaned_audio = np.nan_to_num(
                cleaned_audio,
                nan=0.0,
                posinf=0.0,
                neginf=0.0,
            )

            cleaned_audio = np.clip(
                cleaned_audio,
                -1.0,
                1.0,
            )

            cleaned_rms = self._calculate_rms(
                cleaned_audio
            )

            # Reject suspiciously aggressive processing.
            if cleaned_rms < original_rms * 0.08:
                print(
                    "[NOISE] Suppression rejected: "
                    "excessive audio attenuation."
                )

                return original_audio

            reduction = (
                1.0 - cleaned_rms / original_rms
            ) * 100

            print(
                f"[NOISE] Suppression completed | "
                f"RMS change: {reduction:.1f}%"
            )

            return cleaned_audio

        except Exception as error:

            print(
                f"[NOISE] Suppression failed: {error}"
            )

            print(
                "[NOISE] Using original audio."
            )

            return original_audio

class SpeechToText:
    def __init__(
        self,
        sample_rate: int = 16000,
        channels: int = 1,
        silence_duration: float = 1.5,
        max_recording_seconds: float = 20.0,
        vad_aggressiveness: int = 2,
    ):
        self.audio_recorder = AudioRecorder(
            sample_rate=sample_rate,
            channels=channels,
            silence_duration=silence_duration,
            max_recording_seconds=max_recording_seconds,
            vad_aggressiveness=vad_aggressiveness,
        )

        self.noise_suppressor = NoiseSuppressor(
            sample_rate=sample_rate,
        )
        self.adaptive_engine = AdaptiveVoiceEngine()
        self.model_manager = WhisperModelManager()
        self.task_classifier = VoiceTaskClassifier()

    def _get_model(self):
        decision = self.adaptive_engine.get_decision(
        "conversation",
        gpu_model_loaded=self.model_manager.is_gpu_model_loaded(),
        )

        if (
            decision.device == "cuda"
            and decision.model_size == "medium"
        ):
            model = self.model_manager.get_gpu_model()
        else:
            model = self.model_manager.get_cpu_model()

        return model, decision

    def transcribe_file(self, audio_path: str):
        model, decision = self._get_model()

        print(
            f"[STT] {decision.model_size} "
            f"{decision.device}"
        )

        segments, info = model.transcribe(
            audio_path,
            beam_size=5,
            language=None,
            vad_filter=True,
            vad_parameters={
                "min_silence_duration_ms": 500,
            },
            condition_on_previous_text=False,
        )

        text_parts = []

        for segment in segments:
            text = segment.text.strip()

            if text:
                text_parts.append(text)

        text = " ".join(text_parts).strip()

        if not text:
            return None

        classification = self.task_classifier.classify(text)

        return {
            "text": text,
            "language": info.language,
            "language_probability": info.language_probability,
            "model": decision.model_size,
            "device": decision.device,
            "compute_type": decision.compute_type,
            "reason": decision.reason,
            "task_type": classification.task_type,
            "classification_confidence": classification.confidence,
        }

    def listen(self):
        audio = self.audio_recorder.record_until_silence()

        if audio is None:
            print("No speech detected.")
            return None

        audio = self.noise_suppressor.process(audio)
        temporary_path = self.audio_recorder.save_wav(audio)

        try:
            result = self.transcribe_file(temporary_path)

            if result is None:
                print("No speech recognized.")
                return None

            return result

        finally:
            if os.path.exists(temporary_path):
                os.remove(temporary_path)

from dataclasses import dataclass
import re


@dataclass
class TaskClassification:
    task_type: str
    confidence: float
    reason: str


class VoiceTaskClassifier:
    """
    Classifies transcribed speech into:

        command
        conversation

    This is intentionally lightweight.

    It does NOT use an LLM or GPU.
    """

    COMMAND_PATTERNS = [
        r"^open\b",
        r"^close\b",
        r"^launch\b",
        r"^start\b",
        r"^stop\b",
        r"^play\b",
        r"^pause\b",
        r"^resume\b",
        r"^turn on\b",
        r"^turn off\b",
        r"^enable\b",
        r"^disable\b",
        r"^increase\b",
        r"^decrease\b",
        r"^raise\b",
        r"^lower\b",
        r"^mute\b",
        r"^unmute\b",
        r"^set\b",
        r"^restart\b",
        r"^shutdown\b",
        r"^lock\b",
        r"^search\b",
        r"^go to\b",
        r"^show\b",
        r"^check\b",
    ]

    CONVERSATION_PATTERNS = [
        r"\bwhy\b",
        r"\bhow\b",
        r"\bwhat\b",
        r"\bwhen\b",
        r"\bwhere\b",
        r"\bwho\b",
        r"\bexplain\b",
        r"\btell me\b",
        r"\bcan you explain\b",
        r"\bdo you think\b",
        r"\bwhat do you think\b",
    ]

    def classify(
        self,
        text: str,
    ) -> TaskClassification:

        if not text:
            return TaskClassification(
                task_type="unknown",
                confidence=0.0,
                reason="Empty transcription",
            )

        text = text.strip().lower()

        # Remove excessive whitespace.
        text = re.sub(
            r"\s+",
            " ",
            text,
        )

        # -----------------------------------------
        # COMMAND DETECTION
        # -----------------------------------------

        for pattern in self.COMMAND_PATTERNS:

            if re.search(
                pattern,
                text,
            ):
                return TaskClassification(
                    task_type="command",
                    confidence=0.90,
                    reason=(
                        "Command pattern detected"
                    ),
                )

        # -----------------------------------------
        # CONVERSATION DETECTION
        # -----------------------------------------

        for pattern in self.CONVERSATION_PATTERNS:

            if re.search(
                pattern,
                text,
            ):
                return TaskClassification(
                    task_type="conversation",
                    confidence=0.90,
                    reason=(
                        "Conversation pattern detected"
                    ),
                )

        # -----------------------------------------
        # SHORT UTTERANCE HEURISTIC
        # -----------------------------------------

        words = text.split()

        if len(words) <= 5:

            return TaskClassification(
                task_type="command",
                confidence=0.60,
                reason=(
                    "Short utterance with no "
                    "conversation pattern"
                ),
            )

        # -----------------------------------------

========== text_normalizer.py ==========

import re
import html


def normalize_for_speech(text: str) -> str:
    if not text:
        return ""

    text = html.unescape(text)

    # Remove code blocks.
    text = re.sub(r"```[\s\S]*?```", " ", text)

    # Remove inline code formatting, preserve its content.
    text = re.sub(r"`([^`]+)`", r"\1", text)

    # Convert Markdown links to their readable labels.
    text = re.sub(r"\[([^\]]+)\]\([^)]+\)", r"\1", text)

    # Remove Markdown headings.
    text = re.sub(r"^\s{0,3}#{1,6}\s*", "", text, flags=re.MULTILINE)

    # Remove blockquote markers.
    text = re.sub(r"^\s*>\s?", "", text, flags=re.MULTILINE)

    # Remove bullet and numbered-list markers.
    text = re.sub(
        r"^\s*(?:[-*+]|\d+[.)])\s+",
        "",
        text,
        flags=re.MULTILINE,
    )

    # Preserve text inside brackets and parentheses.
    # Remove only the surrounding formatting symbols.
    text = re.sub(r"\*\*(.*?)\*\*", r"\1", text, flags=re.DOTALL)
    text = re.sub(r"__(.*?)__", r"\1", text, flags=re.DOTALL)
    text = re.sub(r"(?<!\*)\*(?!\s)(.*?)\*(?!\*)", r"\1", text, flags=re.DOTALL)
    text = re.sub(r"(?<!_)_(?!\s)(.*?)_(?!_)", r"\1", text, flags=re.DOTALL)

    # Remove HTML tags.
    text = re.sub(r"<[^>]+>", " ", text)

    # Remove decorative symbols that should not be spoken.
    text = re.sub(r"[#*_~`]", "", text)

    # Normalize whitespace.
    text = re.sub(r"[ \t]+", " ", text)
    text = re.sub(r"\n+", " ", text)

    return text.strip()

import queue
import threading
from pathlib import Path

import sounddevice as sd

from piper import PiperVoice
from app.voice.text_normalizer import normalize_for_speech


class TextToSpeech:
    def __init__(
        self,
        voice_path: str = r"D:\Edith\en_US-lessac-medium.onnx",
    ):
        self.voice_path = Path(voice_path)

        if not self.voice_path.exists():
            raise FileNotFoundError(
                f"Piper voice model not found: {self.voice_path}"
            )

        print("[TTS] Loading Piper voice...")

        self.voice = PiperVoice.load(
            model_path=self.voice_path,
            use_cuda=False,
        )

        self.sample_rate = self.voice.config.sample_rate

        self.output_stream = sd.OutputStream(
            samplerate=self.sample_rate,
            channels=1,
            dtype="float32",
            latency="low",
        )

        self.output_stream.start()

        print("[TTS] Piper voice ready.")

    def speak_stream(self, speech_queue):
        audio_queue = queue.Queue(maxsize=8)
        errors = []

        def audio_generator():
            try:
                while True:
                    text = speech_queue.get()

                    try:
                        if text is None:
                            break

                        clean_text = normalize_for_speech(text)

                        if not clean_text:
                            continue

                        for chunk in self.voice.synthesize(clean_text):
                            audio_queue.put(chunk.audio_float_array)

                    finally:
                        speech_queue.task_done()

            except Exception as error:
                errors.append(error)

            finally:
                audio_queue.put(None)

        producer = threading.Thread(
            target=audio_generator,
            daemon=True,
        )

        producer.start()

        while True:
            audio = audio_queue.get()

            if audio is None:
                break

            self.output_stream.write(
                audio.reshape(-1, 1)
            )

        producer.join()

        if errors:
            raise RuntimeError(
                f"Piper synthesis failed: {errors[0]}"
            ) from errors[0]

    def speak(self, text: str):
        if not text or not text.strip():
            return

        speech_queue = queue.Queue()
        speech_queue.put(text)
        speech_queue.put(None)

        self.speak_stream(speech_queue)

    def close(self):
        if self.output_stream:
            self.output_stream.stop()
            self.output_stream.close()


========== vad.py ==========

import webrtcvad


class VoiceActivityDetector:
    """
    Detects human speech in microphone audio.

    VAD does NOT:
        - transcribe speech
        - understand commands
        - generate responses

    It only determines whether an audio frame
    contains speech.
    """

    def __init__(
        self,
        sample_rate: int = 16000,
        aggressiveness: int = 2,
    ):
        self.sample_rate = sample_rate

        if aggressiveness not in (0, 1, 2, 3):
            raise ValueError(
                "Aggressiveness must be 0, 1, 2, or 3."
            )

        self.aggressiveness = aggressiveness

        self.vad = webrtcvad.Vad(
            aggressiveness
        )

    def is_speech(
        self,
        audio_frame: bytes,
    ) -> bool:
        """
        Determine whether an audio frame contains speech.

        WebRTC VAD expects:
            - 16-bit PCM audio
            - mono audio
            - 10, 20, or 30 ms frames
            - supported sample rates
        """

        return self.vad.is_speech(
            audio_frame,
            self.sample_rate,
        )

import time
from collections import deque
from pathlib import Path

import numpy as np
import onnxruntime as ort
import sounddevice as sd

from openwakeword.utils import AudioFeatures


class WakeWordDetector:
    def __init__(
        self,
        wake_word: str = "hey_edith",
        sample_rate: int = 16000,
        chunk_size: int = 16000,
        threshold: float = 0.90,
        min_rms: float = 30.0,
    ):
        self.wake_word = wake_word
        self.sample_rate = sample_rate
        self.chunk_size = chunk_size
        self.threshold = threshold
        self.min_rms = min_rms

        self.window_seconds = 2
        self.window_size = self.sample_rate * self.window_seconds

        self.cooldown_seconds = 2.0
        self.last_detection_time = 0.0

        # Resolve model paths relative to the EDITH project root.
        project_root = Path(__file__).resolve().parents[2]
        model_dir = project_root / "models" / "wakeword"

        wake_model_path = model_dir / "hey_edith.onnx"
        melspec_path = model_dir / "melspectrogram.onnx"
        embedding_path = model_dir / "embedding_model.onnx"

        required_files = [
            wake_model_path,
            melspec_path,
            embedding_path,
        ]

        for file_path in required_files:
            if not file_path.is_file():
                raise FileNotFoundError(
                    f"Required wake-word model not found: {file_path}"
                )

        print("[WAKE] Loading custom EDITH wake-word model...")

        self.session = ort.InferenceSession(
            str(wake_model_path),
            providers=["CPUExecutionProvider"],
        )

        self.input_name = self.session.get_inputs()[0].name

        self.features = AudioFeatures(
            melspec_model_path=str(melspec_path),
            embedding_model_path=str(embedding_path),
        )

        self.audio_buffer = deque(maxlen=self.window_size)

        print("[WAKE] Custom EDITH wake-word model loaded.")
        print(
            f"[WAKE] Threshold={self.threshold:.2f} | "
            f"Minimum RMS={self.min_rms:.1f}"
        )

    def wait_for_wake_word(self) -> bool:
        self.audio_buffer.clear()

        with sd.InputStream(
            samplerate=self.sample_rate,
            channels=1,
            dtype="int16",
            blocksize=self.chunk_size,
        ) as stream:

            while True:
                audio_data, overflowed = stream.read(self.chunk_size)

                if overflowed:
                    print(
                        "\n[WAKE] Warning: microphone buffer overflow."
                    )

                chunk = np.asarray(
                    audio_data,
                    dtype=np.int16,
                ).flatten()

                self.audio_buffer.extend(chunk.tolist())

                if len(self.audio_buffer) < self.window_size:
                    continue

                audio = np.asarray(
                    self.audio_buffer,
                    dtype=np.int16,
                )

                # Calculate RMS before running expensive model inference.
                audio_float = audio.astype(np.float32)
                rms = float(
                    np.sqrt(np.mean(np.square(audio_float)))
                )

                # Reject near-silence and very low-level background noise.
                if rms < self.min_rms:
                    continue

                clip = audio[np.newaxis, :]

                embedded = self.features.embed_clips(
                    clip,
                    batch_size=1,
                )

                embedded = np.asarray(
                    embedded,
                    dtype=np.float32,
                )

                if embedded.ndim != 3 or embedded.shape[1] < 16:
                    continue

                n = embedded.shape[1]
                start = max(0, (n - 16) // 2)

                model_input = embedded[
                    :,
                    start:start + 16,
                    :,
                ]

                if model_input.shape[1] != 16:
                    continue

                raw_output = self.session.run(
                    None,
                    {self.input_name: model_input},
                )[0]

                output_array = np.asarray(
                    raw_output,
                    dtype=np.float32,
                ).flatten()

                if output_array.size == 0:
                    continue

                score = float(output_array[0])

                if not np.isfinite(score):
                    continue

                print(
                    f"\r[WAKE DEBUG] Score={score:.4f} | "
                    f"RMS={rms:.2f}",
                    end="",
                    flush=True,
                )

                now = time.monotonic()

                # Single-prediction activation: no extra confirmation delay.
                if score >= self.threshold:
                    if (
                        now - self.last_detection_time
                        >= self.cooldown_seconds
                    ):
                        print(
                            f"\n[WAKE] Hey EDITH detected! "
                            f"Score={score:.4f} | RMS={rms:.2f}"
                        )

                        self.last_detection_time = now
                        return True
